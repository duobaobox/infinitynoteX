## 三、实现方案

### 3.1 方案一：主进程直接监听（推荐 ⭐）

在 `main.ts` 中添加存储事件监听，直接触发重索引。

**优点**：

- 实现简单，逻辑集中
- 不依赖渲染进程状态
- 应用启动即生效

**缺点**：

- 需要在主进程中引入防抖逻辑

#### 实现代码

```typescript
// electron/main.ts 或新建 electron/knowledge/realTimeIndexer.ts

import { debounce } from 'lodash';
import { reindexNote, deleteNoteFromIndex, readKnowledgeConfig } from './knowledge/knowledgeIndex';

// 防抖延迟（毫秒）：用户停止编辑后触发
const DEBOUNCE_DELAY = 2000;

// 存储待处理的笔记 ID
const pendingReindex = new Map<string, NodeJS.Timeout>();

/**
 * 初始化实时索引监听器
 * 在 app.whenReady() 之后调用
 */
export function initRealTimeIndexer(): void {
  // 监听 StorageManager 的事件
  // 由于 storageEvents 是通过 IPC 发送的，我们需要在主进程中直接监听
  // 方案：创建一个内部事件总线
}

/**
 * 处理笔记变更事件
 */
async function handleNoteChange(type: 'created' | 'updated', noteId: string): Promise<void> {
  // 检查知识库是否启用
  const config = await readKnowledgeConfig();
  if (!config?.enabled || !config.embedding) {
    return;
  }

  // 取消之前的防抖定时器
  const existingTimer = pendingReindex.get(noteId);
  if (existingTimer) {
    clearTimeout(existingTimer);
  }

  // 设置新的防抖定时器
  const timer = setTimeout(async () => {
    try {
      console.log(`[RealTimeIndexer] Reindexing note: ${noteId}`);
      const result = await reindexNote(noteId);
      if (result.success) {
        console.log(`[RealTimeIndexer] Indexed ${result.vectorCount} vectors for note ${noteId}`);
      } else {
        console.warn(`[RealTimeIndexer] Failed to index note ${noteId}: ${result.error}`);
      }
    } catch (error) {
      console.error(`[RealTimeIndexer] Error indexing note ${noteId}:`, error);
    } finally {
      pendingReindex.delete(noteId);
    }
  }, DEBOUNCE_DELAY);

  pendingReindex.set(noteId, timer);
}

/**
 * 处理笔记删除事件
 */
function handleNoteDelete(noteId: string): void {
  // 取消待处理的重索引
  const existingTimer = pendingReindex.get(noteId);
  if (existingTimer) {
    clearTimeout(existingTimer);
    pendingReindex.delete(noteId);
  }

  // 立即删除索引
  try {
    const deleted = deleteNoteFromIndex(noteId);
    console.log(`[RealTimeIndexer] Deleted ${deleted} vectors for note ${noteId}`);
  } catch (error) {
    console.error(`[RealTimeIndexer] Error deleting index for note ${noteId}:`, error);
  }
}
```

### Step 2: 创建实时索引服务

**新建文件**：`electron/knowledge/realTimeIndexer.ts`

```typescript
/**
 * 实时索引服务
 * 监听笔记变更事件，自动更新知识库索引
 */

import { BrowserWindow, ipcMain } from 'electron';
import { reindexNote, deleteNoteFromIndex, readKnowledgeConfig } from './knowledgeIndex';
import type { StorageEvent } from '../storage/storageEvents';

// 配置
const CONFIG = {
  /** 防抖延迟（毫秒）：用户停止编辑后等待时间 */
  DEBOUNCE_DELAY: 2000,
  /** 最小内容长度：低于此长度不索引 */
  MIN_CONTENT_LENGTH: 10,
  /** 最大并发索引数 */
  MAX_CONCURRENT: 3,
};

// 状态管理
const pendingReindex = new Map<string, NodeJS.Timeout>();
let currentConcurrent = 0;
const indexQueue: string[] = [];

/**
 * 初始化实时索引服务
 * 应在 app.whenReady() 之后调用
 */
export function initRealTimeIndexer(): void {
  console.log('[RealTimeIndexer] Initializing real-time indexer...');

  // 监听主进程内部事件（通过 IPC 桥接）
  ipcMain.on('internal:storage:event', (_, event: StorageEvent) => {
    handleStorageEvent(event);
  });

  console.log('[RealTimeIndexer] Real-time indexer initialized');
}

/**
 * 手动触发存储事件（供 StorageManager 调用）
 */
export function emitInternalStorageEvent(event: StorageEvent): void {
  // 发送到渲染进程
  BrowserWindow.getAllWindows().forEach((win) => {
    if (!win.isDestroyed()) {
      win.webContents.send('storage:event', event);
    }
  });

  // 同时在主进程内部处理
  handleStorageEvent(event);
}

/**
 * 处理存储事件
 */
async function handleStorageEvent(event: StorageEvent): Promise<void> {
  // 只处理笔记相关事件
  if (event.entity !== 'note') {
    return;
  }

  // 检查知识库是否启用
  const config = await readKnowledgeConfig();
  if (!config?.enabled || !config.embedding) {
    return;
  }

  switch (event.type) {
    case 'created':
    case 'updated':
      scheduleReindex(event.id);
      break;
    case 'deleted':
      handleNoteDelete(event.id);
      break;
  }
}

/**
 * 调度重索引（带防抖）
 */
function scheduleReindex(noteId: string): void {
  // 取消之前的定时器
  const existingTimer = pendingReindex.get(noteId);
  if (existingTimer) {
    clearTimeout(existingTimer);
  }

  // 设置新的防抖定时器
  const timer = setTimeout(() => {
    pendingReindex.delete(noteId);
    enqueueReindex(noteId);
  }, CONFIG.DEBOUNCE_DELAY);

  pendingReindex.set(noteId, timer);
}

/**
 * 加入索引队列
 */
function enqueueReindex(noteId: string): void {
  // 避免重复加入队列
  if (indexQueue.includes(noteId)) {
    return;
  }

  indexQueue.push(noteId);
  processQueue();
}

/**
 * 处理索引队列
 */
async function processQueue(): Promise<void> {
  // 并发控制
  if (currentConcurrent >= CONFIG.MAX_CONCURRENT) {
    return;
  }

  const noteId = indexQueue.shift();
  if (!noteId) {
    return;
  }

  currentConcurrent++;

  try {
    console.log(`[RealTimeIndexer] Reindexing note: ${noteId}`);
    const startTime = Date.now();

    const result = await reindexNote(noteId);

    const duration = Date.now() - startTime;
    if (result.success) {
      console.log(
        `[RealTimeIndexer] ✓ Indexed ${result.vectorCount} vectors for note ${noteId} (${duration}ms)`,
      );
    } else {
      console.warn(`[RealTimeIndexer] ✗ Failed to index note ${noteId}: ${result.error}`);
    }
  } catch (error) {
    console.error(`[RealTimeIndexer] Error indexing note ${noteId}:`, error);
  } finally {
    currentConcurrent--;
    // 继续处理队列
    if (indexQueue.length > 0) {
      processQueue();
    }
  }
}

/**
 * 处理笔记删除
 */
function handleNoteDelete(noteId: string): void {
  // 取消待处理的重索引
  const existingTimer = pendingReindex.get(noteId);
  if (existingTimer) {
    clearTimeout(existingTimer);
    pendingReindex.delete(noteId);
  }

  // 从队列中移除
  const queueIndex = indexQueue.indexOf(noteId);
  if (queueIndex !== -1) {
    indexQueue.splice(queueIndex, 1);
  }

  // 立即删除索引
  try {
    const deleted = deleteNoteFromIndex(noteId);
    if (deleted > 0) {
      console.log(`[RealTimeIndexer] Deleted ${deleted} vectors for note ${noteId}`);
    }
  } catch (error) {
    console.error(`[RealTimeIndexer] Error deleting index for note ${noteId}:`, error);
  }
}

/**
 * 获取实时索引器状态
 */
export function getRealTimeIndexerStatus(): {
  pendingCount: number;
  queueLength: number;
  currentConcurrent: number;
} {
  return {
    pendingCount: pendingReindex.size,
    queueLength: indexQueue.length,
    currentConcurrent,
  };
}

/**
 * 清空所有待处理任务
 */
export function clearPendingTasks(): void {
  for (const timer of pendingReindex.values()) {
    clearTimeout(timer);
  }
  pendingReindex.clear();
  indexQueue.length = 0;
}
```

## 五、注意事项与风险控制

### 5.1 性能考虑

| 风险                        | 影响                   | 缓解措施                       |
| --------------------------- | ---------------------- | ------------------------------ |
| **频繁 Embedding API 调用** | 触发速率限制，产生费用 | 2秒防抖，合并连续编辑          |
| **用户快速连续编辑**        | API 调用堆积           | 防抖 + 队列管理                |
| **大笔记处理耗时**          | 阻塞其他操作           | 异步处理 + 进度反馈            |
| **并发索引过多**            | 内存占用高             | 限制并发数（MAX_CONCURRENT=3） |

### 5.2 防抖时间选择

```
用户编辑行为分析：
- 打字间隔：通常 < 500ms
- 段落思考：通常 1-3 秒
- 长时间暂停：> 5 秒

推荐防抖延迟：2000ms（2秒）
- 足够过滤连续打字
- 不会让用户感觉延迟过长
```

### 5.3 错误处理策略

```typescript
// 错误处理等级
enum IndexErrorLevel {
  IGNORE, // 忽略（如：内容太短）
  RETRY, // 重试（如：临时网络问题）
  SKIP, // 跳过（如：笔记已删除）
  ALERT, // 告警（如：API Key 失效）
}

// 错误分类处理
function classifyError(error: Error): IndexErrorLevel {
  const msg = error.message.toLowerCase();

  if (msg.includes('rate limit') || msg.includes('429')) {
    return IndexErrorLevel.RETRY;
  }
  if (msg.includes('unauthorized') || msg.includes('401')) {
    return IndexErrorLevel.ALERT;
  }
  if (msg.includes('not found')) {
    return IndexErrorLevel.SKIP;
  }

  return IndexErrorLevel.IGNORE;
}
```

### 5.4 用户体验考虑

1. **索引状态可见性**
   - 在 UI 中显示索引状态图标（同步中/已索引/失败）
   - 知识库设置页显示实时索引队列状态

2. **手动控制**
   - 保留"重建索引"按钮作为备选
   - 提供"暂停实时索引"选项

3. **错误通知**
   - API 配置错误时弹出提示
   - 连续失败时建议检查网络
